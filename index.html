async function toggleLive() { const titulo = document.getElementById('liveTitleInput').value; const videoElement = document.getElementById('liveVideo'); const placeholder = document.getElementById('livePlaceholder'); const badge = document.getElementById('liveBadge'); const btn = document.getElementById('liveActionBtn'); if (!streamAtivo) { if(!titulo.trim()) { alert("Por favor, insira um título para a sua live antes de começar."); return; } // Pergunta ao utilizador se quer transmitir a câmara ou o ecrã const querEspelhar = confirm("Queres partilhar a tua Tela (Espelhar) ou usar a Câmara?\n\nClica em [OK] para partilhar a Tela ou [Cancelar] para usar a Câmara."); try { if (querEspelhar) { // Pede o fluxo de captura do ecrã/janela streamAtivo = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: true }); } else { // Pede o fluxo normal da câmara streamAtivo = await navigator.mediaDevices.getUserMedia({ video: true, audio: true }); } videoElement.srcObject = streamAtivo; videoElement.style.display = 'block'; placeholder.style.display = 'none'; badge.style.display = 'block'; btn.textContent = "Encerrar Transmissão"; btn.style.background = "linear-gradient(135deg, #475569, #334155)"; // Se o utilizador clicar no botão "Parar Partilha" nativo do navegador streamAtivo.getVideoTracks()[0].onended = function() { fecharTransmissao(); }; } catch (error) { alert("A transmissão foi cancelada ou não foi permitida."); console.error(error); } } else { fecharTransmissao(); } } function fecharTransmissao() { const videoElement = document.getElementById('liveVideo'); const placeholder = document.getElementById('livePlaceholder'); const badge = document.getElementById('liveBadge'); const btn = document.getElementById('liveActionBtn'); if (streamAtivo) { streamAtivo.getTracks().forEach(track => track.stop()); streamAtivo = null; } videoElement.srcObject = null; videoElement.style.display = 'none'; placeholder.style.display = 'block'; badge.style.display = 'none'; btn.textContent = "Iniciar Transmissão"; btn.style.background = "linear-gradient(135deg, var(--live-color), #dc2626)"; }
